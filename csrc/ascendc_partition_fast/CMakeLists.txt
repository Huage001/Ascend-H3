cmake_minimum_required(VERSION 3.16)
project(spark_partition_fast LANGUAGES C CXX)
set(LINUX TRUE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE RELEASE)
set(SOC_VERSION Ascend910B3 CACHE STRING "Target")
include("${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
ascendc_library(preproc_kernels STATIC partition.cpp)
add_library(spark_partition_fast SHARED host.cpp)
target_compile_options(spark_partition_fast PRIVATE -Wno-deprecated-declarations)
target_compile_definitions(spark_partition_fast PRIVATE _GLIBCXX_USE_CXX11_ABI=${GLIBCXX_USE_CXX11_ABI})
target_include_directories(spark_partition_fast PRIVATE "${TORCH_NPU_PATH}/include" "${TORCH_PATH}/include" "${TORCH_PATH}/include/torch/csrc/api/include")
target_link_directories(spark_partition_fast PRIVATE "${TORCH_PATH}/lib" "${TORCH_NPU_PATH}/lib")
target_link_libraries(spark_partition_fast PRIVATE preproc_kernels torch_npu ascendcl tiling_api register platform ascendalog dl)
